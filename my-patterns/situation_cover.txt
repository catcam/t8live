// Yazoo -- "Situation" (1982) cover for T-8 + S-1
// Rebuilt from a REAL MIDI transcription (YAZOO_-_Situation.mid, Nikša's
// Desktop, 2026-08-02) after web research + OpenRouter reconstruction both
// turned out to guess wrong about the actual arrangement. Read via mido:
// tempo 512821 us/beat = exactly 117.0 BPM, 480 ticks/beat, type 1, 19
// tracks. Two tracks mattered:
//   track 3 "synth bass1" (GM prog 38, range B2-C#3, 1032 notes) -- THE
//     driving bassline everyone remembers.
//   track 1 (unnamed lead, GM prog 62, range C#4-C#5, 833 notes) -- the
//     high melodic hook, doubled an octave down by track 2 (GM prog 81
//     "lead2 sawtooth", constant C#4/B3 pattern).
//
// Real finding that overturned the whole earlier model: the bassline is a
// CONSTANT, UNCHANGING 7-note ostinato -- same riff every single bar for
// the entire 5:47, verified by dumping bars 6-9 AND bars 21-25 (deep into
// a pre-chorus/turnaround region by chroma analysis) and getting byte-
// identical note sequences both times. It does NOT follow the chord chart.
// The actual harmonic motion lives entirely in the lead/hook track, which
// alternates a stable 2-bar A/B shape through the verse/chorus vamp, then
// switches register (E3-F#3-G#3-C#4) for the pre-chorus/turnaround lift.
// This inverts my first two drafts, which put the "follows the chords"
// job on the bass and a static stab on the hook -- backwards.
//
// Second real finding: the bass track's first note-on is at bar 6 (~12s
// in), the lead hook's is at bar 2 (~4s in) -- neither starts at bar 0.
// Nikša's ear-check said the hook is there "literally clean from the
// first second," which is closer to the real ~4s than my first-draft
// assumption of nothing-until-the-chorus, so the hook here starts at bar
// 0 of this arrangement (a small liberty for a tight instrumental loop,
// not literal-transcription purism) while the bass keeps its real,
// unchanging 7-note shape once it's your turn to bring it in.
//
// Drum pattern is still NOT a transcription (the MIDI file has real drum
// tracks split by GM note number across tracks 8-18, but reconciling 8
// overlapping tracks against the T-8's 6 rhythm voices is future work,
// not done tonight) -- this is the same period-accurate TR-808 groove as
// before: four-on-floor kick, driving 8th hats, backbeat clap.

// t8drum/t8bass/T8_PORT/s1note/S1_PORT/note/stack/arrange/silence/setcps
// are all already global in the REPL (evalScope pulls in @strudel/t8 +
// @strudel/s1 alongside core) -- no import here, the editor's transpiler
// rejects import/export syntax.

setcps(117 / 60 / 4);

// real bass ostinato (track 3), constant every bar, steps 0/4/7/9/10/12/14
// -- built with array.join(' '), not a template literal: the REPL
// transpiler auto-wraps ANY backtick string as mini-notation using just
// its raw pre-interpolation text, which breaks on "${...}" sequences
const bassRiff = ['c#3', '~', '~', '~', 'c#3', '~', '~', 'b2', '~', 'b2', 'b2', '~', 'b3', '~', 'b2', '~'].join(' ');

// real hook, verse/chorus shape -- alternates every bar (line A on the
// "C#-feeling" bar, line B on the "B-feeling" bar, matching how track 1
// actually alternates bar-to-bar in the MIDI)
const hookLineA = ['c#4', '~', 'c#4', '~', 'c#5', '~', '~', '~', '~', '~', '~', '~', 'b4', 'g#4', 'b4', 'c#5'].join(' ');
const hookLineB = ['c#4', '~', 'c#4', '~', 'c#5', '~', '~', '~', '~', '~', 'b4', '~', 'e5', 'd#5', 'b4', 'c#5'].join(' ');

// real hook, pre-chorus/turnaround phrase -- a real TWO-bar melodic idea
// (bars 23 and 25 of the MIDI, deep in a chroma-confirmed F#/G#/D#
// region), not one bar repeated. First draft repeated a single bar here,
// which is likely what sounded off on the first real listen -- these are
// the two actual consecutive bars, not a guess.
const hookLiftA = ['c#3', '~', 'c#3', '~', 'e3', '~', 'e3', '~', 'f#3', '~', 'g#3', 'f#3', 'g#3', 'c#4', '~', '~'].join(' ');
const hookLiftB = ['c#3', '~', 'c#3', '~', 'e3', '~', 'e3', '~', 'g#3', '~', 'f#3', 'e3', 'f#3', 'c#3', '~', '~'].join(' ');

stack(
  // T-8: drums, still a reconstruction (see header) -- four-on-floor kick,
  // driving 8th hats, backbeat clap
  t8drum("bd ~ ~ ~ bd ~ ~ ~ bd ~ ~ ~ bd ~ ~ ~").midi(T8_PORT),
  t8drum("ch ~ ch ~ ch ~ ch ~ ch ~ ch ~ ch ~ ch ~").midi(T8_PORT),
  t8drum("~ ~ ~ ~ clap ~ ~ ~ ~ ~ ~ ~ clap ~ ~ ~").midi(T8_PORT),

  // S-1: the real bass ostinato, constant for all 14 bars (matches the
  // real recording -- it never changes, regardless of what's happening
  // harmonically above it)
  note(mini(bassRiff)).s1note().midi(S1_PORT),

  // T-8 bass channel: the real hook, alternating A/B through verse+chorus,
  // switching to the real "lift" shape for pre-chorus + the chorus
  // turnaround (mini() wraps each bar string explicitly -- note() only
  // auto-parses mini-notation for string LITERALS the transpiler can see
  // statically; a runtime-built string in a variable needs an explicit
  // mini() call, otherwise the whole bar string gets treated as one
  // unparsed note name)
  arrange(
    [1, note(mini(hookLineA))], [1, note(mini(hookLineB))], [1, note(mini(hookLineA))], [1, note(mini(hookLineB))], // verse
    [1, note(mini(hookLiftA))], [1, note(mini(hookLiftB))], // pre-chorus
    [1, note(mini(hookLineA))], [1, note(mini(hookLineB))], // chorus line 1
    [1, note(mini(hookLineA))], [1, note(mini(hookLineB))], // chorus line 2
    [1, note(mini(hookLineA))], [1, note(mini(hookLineB))], // chorus line 3
    [1, note(mini(hookLiftA))], [1, note(mini(hookLiftB))], // chorus turnaround
  )
    .t8bass()
    .midi(T8_PORT),
);

// No vocal/lead here on purpose -- Alison Moyet's melody (and the "Move
// out, don't mess around" hook itself) is the one part that genuinely
// can't be covered by two synths. This is instrumental-only.
